import { runBot } from "@neontechspace/fluxerly"

await runBot({
    token: process.env.FLUXER_BOT_TOKEN,
    commands: {
        prefix: "!",
        commands: {
            help: {
                description: "List the commands",
                execute: ({ sendHelp }) => sendHelp(),
            },
            rules: {
                description: "Show the rules, five at a time",
                execute: ({ paginate }) => paginate(["Rules 1 to 5", "Rules 6 to 10"]),
            },
        },
    },
})
